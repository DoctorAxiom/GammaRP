\newcommand{\GRPdie}[2][0]{%
	\IfStrEq{#1}{alt}{%
		\GammaDodec[color=grpalt!30!white]{\color{grpalt}#2}%
	}{%
		\ifnum #1=0%0 case
	 		\ifnum#2=12%
				\GammaDodec[color=grppos!30!white]{\textbf{\color{grppos}#2}}%
			\else%
				\GammaDodec{#2}%
			\fi%
		\fi%
	 	\ifnum#1>0% advantage case
	 		\ifnum#2>\inteval{12-#1}%
				\GammaDodec[color=grppos!30!white]{\textbf{\color{grppos}#2}}%
			\else%
				\GammaDodec{#2}%
			\fi%
		\fi%
		\ifnum#1<0% disadvantage case (when just the 12s should be hilighted yellow)
			\ifnum#2=12%
				\GammaDodec[color=grpmid!30!white]{\color{grpmid}#2}%
			\else%
				\ifnum#2=1%
					\GammaDodec[color=grpneg!30!white]{\color{grpneg}#2}%
				\else%
					\GammaDodec{#2}%
				\fi%
			\fi%
		\fi%
	}%
}

\newcounter{GRPdisadvdummy}
\newcounter{GRPdisadvsuccdum}
\newcounter{GRPtwelves}
\newcommand{\GRPdicelist}[2][0]{
	\setcounter{GRPdisadvdummy}{0}%
	\setcounter{GRPdisadvsuccdum}{0}%
	\setcounter{GRPtwelves}{0}%
	\ifnum#1<0%disadv case
		\foreach \dieval in {#2}{%
			\ifnum\dieval>1%
				\stepcounter{GRPdisadvsuccdum}%
				\ifnum\dieval=12%
					\stepcounter{GRPtwelves}%
				\fi%
			\fi%
		}% at this point GRPdisadvsuccdum holds the number of dice>1
		\foreach \dieval in {#2}{%
			\ifnum\dieval=12%
				\stepcounter{GRPdisadvdummy}%
				\ifnum\inteval{#1+\arabic{GRPdisadvsuccdum}+1}>0%
					\GRPdie{12}%
				\else%
					\GRPdie[#1]{12}%
				\fi%
			\else%
				\ifnum\dieval>1%
					\stepcounter{GRPdisadvdummy}%
					\GRPdie{\dieval}%
				\else%
					\GRPdie[#1]{\dieval}%
				\fi%
			\fi%
			\ifnum\arabic{GRPtwelves}>0%
			\ifnum\dieval>1%
				\ifnum\inteval{#1+\arabic{GRPdisadvdummy}+1}>0%
					\nolinebreak\hspace{-1.4pt}\textsubscript{\raisebox{-1.4pt}{\textbf{\color{grppos}\arabic{GRPdisadvdummy}}}}\hspace{-0.8pt}%
				\else%
					\ifnum\arabic{GRPdisadvdummy}>\inteval{\arabic{GRPdisadvsuccdum}-1}%
						\nolinebreak\hspace{-1.4pt}\textsubscript{\raisebox{-1.4pt}{\textbf{\textit{\color{grpneg}\arabic{GRPdisadvdummy}}}}}\hspace{-0.8pt}%
					\else%
						\nolinebreak\hspace{-1.4pt}\textsubscript{\raisebox{-1.4pt}{\color{grpalt}\arabic{GRPdisadvdummy}}}\hspace{-0.8pt}%
					\fi%
				\fi%
			\fi%
			\fi%
		\allowbreak}%
	\else%advantage case which is way easier to format
		\foreach \dieval in {#2}{%
			\GRPdie[#1]{\dieval}%
		\allowbreak}%
	\fi%
}
