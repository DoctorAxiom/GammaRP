\newcommand{\GRPdie}[2][0]{%
	\ifnum #1=0%0 case
 		\ifnum#2=12%
			\GammaDodec[color=grppos!30!white]{\textbf{\color{grppos}#2}}%
		\else%
			\GammaDodec{#2}%
		\fi%
	\fi%
 	\ifnum#1>0% advantage case
 		\ifnum#2>\inteval{12-#1}%
			\GammaDodec[color=grppos!30!white]{\textbf{\color{grppos}#2}}%
		\else%
			\GammaDodec{#2}%
		\fi%
	\fi
	\ifnum#1<0% disadvantage case (when just the 12s should be hilighted)
		\ifnum#2=12%
			\GammaDodec[color=grpmid!30!white]{\color{grpmid}#2}%
		\else%
			\GammaDodec{#2}%
		\fi
 	\fi%
}

\newcounter{GRPdisadvdummy}
\newcommand{\GRPdicelist}[2][0]{
	\ifnum #1<0%disadv case
		\setcounter{GRPdisadvdummy}{0}
		\foreach \dieval in {#2}{%
			\ifnum \dieval=12%
				\ifnum #1>-12% this ensures that if disadv is >=12 then there are no fully green dice (guarranteed fail)
			  	\stepcounter{GRPdisadvdummy}%
			  \fi%
				\ifnum \value{GRPdisadvdummy}>\inteval{-1-(#1)}%
					\GRPdie{\dieval}%
				\else%
					\GRPdie[#1]{\dieval}%
				\fi%
			\else%
				\GRPdie[#1]{\dieval}%
			\fi%
		}%
	\else
		\foreach \dieval in {#2}{%
			\GRPdie[#1]{\dieval}%
		}%
	\fi
}
